
#include <stdio.h>
#include <omp.h>

int main()
{
    #pragma omp parallel for collapse(2)
    for (int r = 0; r < 5; r++)
    {
        for (int k = 0; k < 10; k++)
        {
            printf("Thread %d: r=%d k=%d\n",
                   omp_get_thread_num(), r, k);
        }
    }

    return 0;
}

#include <stdio.h>
#include <omp.h>

int fib(int n)
{
    if (n < 2)
        return n;

    int x, y;

    #pragma omp task shared(x)
    x = fib(n - 1);

    #pragma omp task shared(y)
    y = fib(n - 2);

    #pragma omp taskwait

    return x + y;
}

int main()
{
    int n = 20;
    int result;

    #pragma omp parallel
    {
        #pragma omp single
        {
            result = fib(n);
        }
    }

    printf("fib(%d) = %d\n", n, result);

    return 0;
}
