
#include <omp.h>
#include <stdio.h>

int main()
{
    #pragma omp parallel
    {
        int th = omp_get_thread_num();

        printf("Before Master - thread # %d\n", th);

        #pragma omp master
        {
            printf("Inside Master - thread # %d\n",
                   omp_get_thread_num());
        }

        printf("After Master - thread # %d\n", th);
    }

    return 0;
}
